# --- Environment Set-up ---

cd ~
unset CMAKE_PREFIX_PATH
unset GeoModel_DIR
source /cvmfs/sft.cern.ch/lcg/views/LCG_105/x86_64-el9-gcc13-opt/setup.sh

mkdir -p /eos/user/e/elwright/geomodel_stack/externals
cd /eos/user/e/elwright/geomodel_stack/externals

# --- Eigen ---

git clone https://gitlab.com/libeigen/eigen.git
cd eigen

mkdir build install
cd build

cmake .. \
  -DCMAKE_INSTALL_PREFIX=../install

make install

# --- Motif ---

echo $DISPLAY (WORKED)

curl -L -o motif.tar.gz https://sourceforge.net/projects/motif/files/Motif%202.3.8%20Source%20Code/motif-2.3.8.tar.gz/download
tar -xvzf motif.tar.gz
cd motif-2.3.8
make clean
./configure   --prefix=/eos/user/e/elwright/geomodel_stack/externals/motif/install   CPPFLAGS="-I/usr/include/freetype2"
make -j2
make install
export EIGEN_ROOT=/eos/user/e/elwright/geomodel_stack/externals/eigen/install
export MOTIF_ROOT=/eos/user/e/elwright/geomodel_stack/externals/motif/install

# --- GeoModel + GeoModelG4 ---

cd /eos/user/e/elwright/geomodel_stack
git clone https://gitlab.cern.ch/GeoModelDev/GeoModel.git
cd GeoModel

mkdir build install
cd build

cmake .. \
  -DCMAKE_INSTALL_PREFIX=../install \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_CXX_STANDARD=20 \
  -DCMAKE_PREFIX_PATH=$CMAKE_PREFIX_PATH

make -j4
make install

cd /eos/user/e/elwright/geomodel_stack
git clone https://gitlab.cern.ch/GeoModelDev/GeoModelG4.git
cd GeoModelG4

mkdir build install
cd build

cmake .. \
  -DCMAKE_INSTALL_PREFIX=../install \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_CXX_STANDARD=20 \
  -DCMAKE_PREFIX_PATH=/eos/user/e/elwright/geomodel_stack/GeoModel/install:$CMAKE_PREFIX_PATH

make -j4
make install

export CMAKE_PREFIX_PATH=/eos/user/e/elwright/geomodel_stack/GeoModel/install:$CMAKE_PREFIX_PATH
export EIGEN_ROOT=/eos/user/e/elwright/geomodel_stack/externals/eigen/install
export MOTIF_ROOT=/eos/user/e/elwright/geomodel_stack/externals/motif/install

cd ~/GeoModelSplitCal
mkdir build
cd build

cmake .. \
  -DCMAKE_PREFIX_PATH=$CMAKE_PREFIX_PATH

make -j4
