# --- Environment Set-up ---

cd ~
unset CMAKE_PREFIX_PATH
unset GeoModel_DIR
source /cvmfs/sft.cern.ch/lcg/views/LCG_108/x86_64-el9-gcc13-opt/setup.sh

mkdir -p /eos/user/e/elwright/geomodel_stack/externals
cd /eos/user/e/elwright/geomodel_stack/externals





# --- Eigen ---

git clone https://gitlab.com/libeigen/eigen.git
cd eigen

mkdir build install
cd build

cmake .. \
  -DCMAKE_INSTALL_PREFIX=../install

make install






# --- Motif ---

echo $DISPLAY (WORKED)

curl -L -o motif.tar.gz https://sourceforge.net/projects/motif/files/Motif%202.3.8%20Source%20Code/motif-2.3.8.tar.gz/download
tar -xvzf motif.tar.gz
cd motif-2.3.8
make clean
./configure   --prefix=/eos/user/e/elwright/geomodel_stack/externals/motif/install   CPPFLAGS="-I/usr/include/freetype2"
make -j2
make install

export EIGEN_ROOT=/eos/user/e/elwright/geomodel_stack/externals/eigen/install
export MOTIF_ROOT=/eos/user/e/elwright/geomodel_stack/externals/motif/install






# --- Qt6 ---

mkdir -p /eos/user/e/elwright/geomodel_stack/externals/qt6
cd /eos/user/e/elwright/geomodel_stack/externals

git clone https://code.qt.io/qt/qt5.git qt6
cd qt6

perl init-repository --module-subset=default

(IF THIS FAILS, RUN
git submodule sync
git submodule update --init --recursive)

mkdir build && cd build

realpath /eos/user/e/elwright/geomodel_stack/externals/qt6/build  (WORKED)
cd $(realpath /eos/user/e/elwright/geomodel_stack/externals/qt6/build)

cmake .. \
  -DCMAKE_INSTALL_PREFIX=/eos/user/e/elwright/geomodel_stack/externals/qt6/install \
  -DCMAKE_BUILD_TYPE=Release \
  -DBUILD_SHARED_LIBS=ON \
  -DQT_FEATURE_opengl=desktop

make -j2
make install






# --- GeoModel + GeoModelG4 ---

cd /eos/user/e/elwright/geomodel_stack
git clone https://gitlab.cern.ch/GeoModelDev/GeoModel.git
cd GeoModel

mkdir build install
cd build

cmake .. \
  -DCMAKE_INSTALL_PREFIX=/eos/user/e/elwright/geomodel_stack/GeoModel/install \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_CXX_STANDARD=20 \
  -DGEOMODEL_BUILD_GEOMODELG4=ON \
  -DGEOMODEL_BUILD_VISUALIZATION=ON \
  -DMOTIF_INCLUDE_DIR=$MOTIF_ROOT/include \
  -DMOTIF_LIBRARIES=$MOTIF_ROOT/lib/libXm.so

make -j2
make install




# --- GeoModelSplitCal ---

cd /eos/user/e/elwright/
git clone https://github.com/ElliottPWright/GeoModelSplitCal.git
cd GeoModelSplitCal
mkdir build
cd build

export CMAKE_PREFIX_PATH=/eos/user/e/elwright/geomodel_stack/GeoModel/install:$CMAKE_PREFIX_PATH

cmake .. \
  -DCMAKE_PREFIX_PATH=$CMAKE_PREFIX_PATH \
  -DMOTIF_INCLUDE_DIR=$MOTIF_ROOT/include \
  -DGeoModelCore_DIR=$GeoModelCore_DIR \
  -DMOTIF_LIBRARIES="$MOTIF_ROOT/lib/libXm.so" \
  -DCMAKE_EXE_LINKER_FLAGS="-lXft"

make -j4
